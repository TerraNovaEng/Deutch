function updateClock() { const now = new Date(); const hours = now.getHours(); const minutes = now.getMinutes(); const seconds = now.getSeconds(); const hoursStr = String(hours).padStart(2, '0'); const minutesStr = String(minutes).padStart(2, '0'); const secondsStr = String(seconds).padStart(2, '0'); const clockEl = document.getElementById('live-clock'); const labelEl = document.getElementById('clock-german-label'); if (clockEl) { clockEl.textContent = `${hoursStr}:${minutesStr}:${secondsStr}`; } if (labelEl) { let h12 = hours % 12; if (h12 === 0) h12 = 12; let hWord = germanHours[h12 % 12]; let nextHWord = germanHours[(h12 + 1) % 12]; // Helper to spell out numbers in German up to 60 function spellNumber(n) { const units = ["null", "eins", "zwei", "drei", "vier", "fünf", "sechs", "sieben", "acht", "neun", "zehn", "elf", "zwölf", "dreizehn", "vierzehn", "fünfzehn", "sechzehn", "siebzehn", "achtzehn", "neunzehn"]; const tens = ["", "", "zwanzig", "dreißig", "vierzig", "fünfzig"]; if (n < 20) return units[n]; let t = Math.floor(n / 10); let u = n % 10; if (u === 0) return tens[t]; return units[u] + "und" + tens[t]; } let minWord = minutes === 1 ? "eine Minute" : spellNumber(minutes) + " Minuten"; let timeText = ""; if (minutes === 0) { timeText = `Es ist ${hWord} Uhr`; } else if (minutes === 15) { timeText = `Es ist viertel nach ${hWord}`; } else if (minutes === 30) { timeText = `Es ist halb ${nextHWord}`; } else if (minutes === 45) { timeText = `Es ist viertel vor ${nextHWord}`; } else if (minutes < 30) { timeText = `Es ist ${minWord} nach ${hWord}`; } else { let rem = 60 - minutes; let remWord = rem === 1 ? "eine Minute" : spellNumber(rem) + " Minuten"; timeText = `Es ist ${remWord} vor ${nextHWord}`; } let timeGreeting = "Guten Tag!"; if (hours >= 5 && hours < 11) timeGreeting = "Guten Morgen!"; else if (hours >= 11 && hours < 15) timeGreeting = "Mittagszeit!"; else if (hours >= 15 && hours < 18) timeGreeting = "Guten Nachmittag!"; else if (hours >= 18 && hours < 22) timeGreeting = "Guten Abend!"; else timeGreeting = "Gute Nacht!"; labelEl.textContent = `${timeGreeting} ${timeText}`; } const secondDeg = (seconds / 60) * 360; const minuteDeg = ((minutes + seconds / 60) / 60) * 360; const hourDeg = (((hours % 12) + minutes / 60) / 12) * 360; const secondHand = document.getElementById('second-hand'); const minuteHand = document.getElementById('minute-hand'); const hourHand = document.getElementById('hour-hand'); if (secondHand) secondHand.style.transform = `translateX(-50%) rotate(${secondDeg}deg)`; if (minuteHand) minuteHand.style.transform = `translateX(-50%) rotate(${minuteDeg}deg)`; if (hourHand) hourHand.style.transform = `translateX(-50%) rotate(${hourDeg}deg)`; }
